model ClientAssignment {
  id             String       @id @default(uuid()) @db.Uuid
  clientId   String       @db.Uuid
  advisorId      String       @db.Uuid
  assignedBy     String       @db.Uuid
  assignedAt     DateTime     @default(now()) @db.Timestamptz(6)
  endedAt        DateTime?    @db.Timestamptz(6)
  isActive       Boolean      @default(true)
  createdAt      DateTime     @default(now()) @db.Timestamptz(6)
  updatedAt      DateTime     @updatedAt @db.Timestamptz(6)
  advisor        Advisor      @relation("Advisor", fields: [advisorId], references: [id], onDelete: Cascade)
  assignedByUser User         @relation("AssignedBy", fields: [assignedBy], references: [id])
  client     Subscription @relation("Client", fields: [clientId], references: [id], onDelete: Cascade)

  @@unique([clientId, advisorId, isActive])
}

model Subscription {
  id                   String                 @id @default(uuid()) @db.Uuid
  userId               String                 @unique @db.Uuid
  plan                 SubscriptionPlan       @default(STARTER)
  status               SubscriptionStatus     @default(TRIALING)
  billingInterval      BillingInterval        @default(month)
  customerId           String?                @db.VarChar(255)
  subscriptionId       String?                @db.VarChar(255)
  startsAt             DateTime?              @db.Timestamptz(6)
  endsAt               DateTime?              @db.Timestamptz(6)
  cancelAtPeriodEnd    Boolean                @default(false)
  canceledAt           DateTime?              @db.Timestamptz(6)
  trialStart           DateTime?              @db.Timestamptz(6)
  trialEnd             DateTime?              @db.Timestamptz(6)
  metadata             Json?
  createdAt            DateTime               @default(now()) @db.Timestamptz(6)
  updatedAt            DateTime               @updatedAt @db.Timestamptz(6)
  user                 User                   @relation(fields: [userId], references: [id], onDelete: Cascade)
  clientAssignment ClientAssignment[] @relation("Client")
}
